5. Custom Cell Component
function CustomCellComponent(cellData, rowIndex, colIndex, headerText, rowData) {
    if (headerText === 'Status') {
        return {
            tag: 'div',
            class: `status-cell status-${cellData.toLowerCase()}`,
            children: [cellData]
        };
    }
    if (headerText === 'Progress') {
        return {
            tag: 'div',
            class: 'progress-bar',
            children: [{
                tag: 'div',
                class: 'progress-fill',
                style: { width: `${cellData}%` }
            }]
        };
    }
    return { tag: 'span', children: [cellData] };
}
6. Sortable Table with Custom Header Components
// Custom sortable header component
function SortableHeaderComponent(headerData, colIndex, table) {
    return {
        tag: 'div',
        class: 'clickable-header',
        children: [
            headerData,
            { tag: 'span', class: 'sort-indicator', children: ['⇅'] }
        ],
        onClick: function() {
            // Toggle sort direction
            const currentSort = table.currentSort || {};
            let direction = 'asc';
            
            if (currentSort.column === colIndex) {
                direction = currentSort.direction === 'asc' ? 'desc' : 'asc';
            }
            
            table.sortByColumn(colIndex, direction);
            table.currentSort = { column: colIndex, direction };
        }
    };
}
